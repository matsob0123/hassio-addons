# Root only prepares app paths, then Python and Squid run as UID/GID 10001.
# No NET_ADMIN, host namespaces, device access, API tokens or raw sockets.
#include <tunables/global>
profile squid_proxy flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  signal (send, receive),
  # docker-init forwards stop signals across the bootstrap UID change.
  capability kill,
  capability chown,
  capability fowner,
  capability dac_override,
  capability setuid,
  capability setgid,
  /bin/busybox ixr,
  /bin/sh ixr,
  /sbin/docker-init ixr,
  /dev/init ixr,
  /app/run.sh ixr,
  /usr/bin/python3* ixr,
  /usr/sbin/squid ixr,
  /usr/lib/squid/basic_ncsa_auth ixr,
  /usr/bin/htpasswd ixr,
  /usr/bin/openssl ixr,
  /app/** r,
  /etc/** r,
  /usr/lib/** mr,
  /lib/** mr,
  /usr/share/errors/** r,
  /usr/share/icons/** r,
  /usr/share/zoneinfo/** r,
  /data/ rw,
  /data/** rwk,
  /run/squid-ha/ rw,
  /run/squid-ha/** rwk,
  /ssl/ r,
  /ssl/** r,
  /tmp/ rw,
  /tmp/** rwk,
  /dev/shm/ r,
  /dev/shm/squid-* rwk,
  /dev/null rw,
  /dev/random r,
  /dev/urandom r,
  /dev/stdout rw,
  /dev/stderr rw,
  /proc/ r,
  /proc/*/ r,
  /proc/*/stat r,
  /proc/*/status r,
  /proc/*/maps r,
  /proc/*/fd/ r,
  /proc/*/fd/* rw,
  /proc/*/attr/current r,
  /proc/meminfo r,
  /proc/stat r,
  /proc/cpuinfo r,
  /sys/fs/cgroup/** r,
}
